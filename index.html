function doPost(e) {
  try {
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    var data = JSON.parse(e.postData.contents || "{}");

    if (data.action === "sendOtp") {
      MailApp.sendEmail(data.email, "Password Reset OTP", "Your verification code is: " + data.otp);
      return ContentService.createTextOutput(JSON.stringify({success: true}))
        .setMimeType(ContentService.MimeType.JSON);
    }

    if (!data.date || !data.amount || !data.expenseBy) {
      throw new Error("Date, amount and expense person are required.");
    }

    // Keep the expense date as an explicit yyyy-MM-dd string. This prevents
    // Google Sheets / Apps Script timezone conversion from moving 01-Oct to 30-Sep.
    var dateText = String(data.date).trim().substring(0, 10);
    var amount = Number(data.amount);
    if (!/^\d{4}-\d{2}-\d{2}$/.test(dateText) || !isFinite(amount)) {
      throw new Error("Invalid expense date or amount.");
    }

    sheet.appendRow([dateText, amount, String(data.expenseBy).trim(), String(data.remarks || "").trim()]);
    SpreadsheetApp.flush();

    return ContentService.createTextOutput(JSON.stringify({success: true}))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({success: false, error: String(err.message || err)}))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

function doGet(e) {
  try {
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    // getDisplayValues() returns the date exactly as displayed in the Sheet,
    // avoiding UTC serialization of Date objects.
    var rows = sheet.getDataRange().getDisplayValues();
    var data = [];

    for (var i = 1; i < rows.length; i++) {
      var row = rows[i];
      if (!row[0] && !row[1] && !row[2] && !row[3]) continue;
      data.push({
        date: String(row[0] || "").trim(),
        amount: row[1] || "",
        expenseBy: String(row[2] || "").trim(),
        remarks: String(row[3] || "").trim()
      });
    }

    return ContentService.createTextOutput(JSON.stringify({success: true, data: data}))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({success: false, error: String(err.message || err)}))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

Help & Setup Guide

Connecting Google Apps Script

  1. Open your Google Sheet and go to Extensions > Apps Script.
  2. Paste the doGet and doPost functions provided in the Apps Script tab.
  3. Click Deploy > New deployment (Select Web app, set access to Anyone).
  4. Copy the Web App URL and paste it into Admin Panel > Apps Script & Setup.
